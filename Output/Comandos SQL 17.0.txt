CREATE TABLE IF NOT EXISTS customers (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(50),
    telefono VARCHAR(20),
    tipo_dispositivo VARCHAR(50),
    marca VARCHAR(50),
    modelo VARCHAR(50),
    estatus VARCHAR(20),
    fecha_entregar DATETIME,
    motivo VARCHAR (100),
    persona_recibio VARCHAR(50),
    fecha_recibido DATETIME,
    persona_reparo VARCHAR(50),
    diagnostico VARCHAR (500),
    fecha_reparado DATETIME,
    costo INT
);


CREATE TABLE IF NOT EXISTS person (
	id INT AUTO_INCREMENT PRIMARY KEY,
	nombre VARCHAR(50),
	telefono VARCHAR(50),
	salario DECIMAL(10, 2),
	lunes DECIMAL(10, 2),
	martes DECIMAL(10, 2),
	miercoles DECIMAL(10, 2),
	jueves DECIMAL(10, 2),
	viernes DECIMAL(10, 2),
	sabado DECIMAL(10, 2),
	domingo DECIMAL(10, 2)
);


CREATE TABLE IF NOT EXISTS report (
    id INT AUTO_INCREMENT PRIMARY KEY,
    fechaInicio DATE,
    fechaFin DATE,
    ingresoTotal DECIMAL(10, 2),
    salarios DECIMAL(10, 2),
    manoDeObra DECIMAL(10, 2),
    ganancia DECIMAL(10, 2),
    refacciones DECIMAL (10, 2)
);


SET GLOBAL event_scheduler = ON;

SELECT * FROM information_schema.EVENTS WHERE EVENT_NAME = 'update_status_to_atrasado';



DELIMITER //

CREATE EVENT IF NOT EXISTS update_status_to_atrasado
ON SCHEDULE EVERY 1 MINUTE
DO
BEGIN
    UPDATE customers
    SET estatus = 'ATRASADO'
    WHERE estatus = 'PENDIENTE' AND NOW() > fecha_entregar;
END //

DELIMITER ;












DELIMITER //

CREATE EVENT IF NOT EXISTS create_weekly_report
ON SCHEDULE EVERY 1 WEEK
STARTS '2024-09-16 00:00:00' -- El próximo lunes que empieza el evento
DO
BEGIN
    DECLARE fechaInicio DATE;
    DECLARE fechaFin DATE;
    
    -- Calcular la fecha de inicio (lunes de la semana actual)
    SET fechaInicio = CURRENT_DATE() - INTERVAL (DAYOFWEEK(CURRENT_DATE()) - 2) DAY;

    -- Calcular la fecha de fin (domingo de la misma semana)
    SET fechaFin = fechaInicio + INTERVAL 6 DAY;

    -- Insertar los valores en la tabla report con valores iniciales de 0.00
    INSERT INTO report (fechaInicio, fechaFin, ingresoTotal, salarios, ganancia)
    VALUES (fechaInicio, fechaFin, 0.00, 0.00, 0.00);

    -- Restablecer los salarios a 0.00 en todos los registros de la tabla person
    UPDATE person
    SET salario = 0.00;
    
END //

DELIMITER ;
